// runStage.ts — the heart. Does ONE stage. Never decides what runs next.
export async function runStage(input: StageInput): Promise<StageResult> {
  const start = Date.now();
  if (!input.ticketId) {
    throw new Error("missing ticketId");
  }
  const result = await reason(input);
  return { ...result, ms: Date.now() - start };
}
